// Tourist Mobile App Component
const TouristApp = ({ screen, setScreen, language, setLanguage, sosSent, setSosSent, arActive, setArActive }) => {

    case 'registration':
      return <RegistrationScreen setScreen={setScreen} />;
    case 'home':
      return <HomeScreen setScreen={setScreen} language={language} setLanguage={setLanguage} sosSent={sosSent} setSosSent={setSosSent} />;
    case 'safetyAlerts':
      return <SafetyAlertsScreen setScreen={setScreen} />;
    case 'geoFencing':
      return <GeoFencingScreen setScreen={setScreen} />;
    case 'incidentResponse':
      return <IncidentResponseScreen setScreen={setScreen} sosSent={sosSent} setSosSent={setSosSent} />;
    case 'digitalId':
      return <DigitalIdScreen setScreen={setScreen} />;
    case 'aiAssistant':
      return <AIAssistantScreen setScreen={setScreen} language={language} setLanguage={setLanguage} />;
    case 'iotIntegration':
      return <IotIntegrationScreen setScreen={setScreen} />;
    case 'arSafeRoute':
      return <ArSafeRouteScreen setScreen={setScreen} arActive={arActive} setArActive={setArActive} />;
    default:
      return <WelcomeScreen setScreen={setScreen} />;
  }
};

const AIAssistantScreen = ({ setScreen, language, setLanguage }) => {
  const [messages, setMessages] = useState([
    { id: 1, text: "Hello! I'm your Tourist Guardian AI. How can I help you today?", sender: 'ai' }
  ]);
  const [inputText, setInputText] = useState('');
  const [isListening, setIsListening] = useState(false);
  
  const handleSend = () => {
    if (inputText.trim() === '') return;
    
    // Add user message
    setMessages(prev => [...prev, { id: Date.now(), text: inputText, sender: 'user' }]);
    
    // Simulate AI response
    setTimeout(() => {
      let response = "";
      if (inputText.toLowerCase().includes('hospital')) {
        response = "The nearest hospital is Hôpital Européen Georges-Pompidou, 500m away. Would you like directions?";
      } else if (inputText.toLowerCase().includes('police')) {
        response = "The nearest police station is Commissariat du 7e Arrondissement, 800m away. Would you like me to alert them?";
      } else if (inputText.toLowerCase().includes('restaurant')) {
        response = "I recommend Le Jules Verne for fine dining or Les Deux Magots for traditional French cuisine. Both are within 1km.";
      } else {
        response = "I'm here to help with safety and travel information. You can ask me about nearby services, directions, or emergency assistance.";
      }
      
      setMessages(prev => [...prev, { id: Date.now(), text: response, sender: 'ai' }]);
    }, 1000);
    
    setInputText('');
  };
  
  const toggleListening = () => {
    setIsListening(!isListening);
    // In a real app, this would trigger speech recognition
  };
  
  return (
    <div className="h-full flex flex-col">
      <div className="flex justify-between items-center mb-4">
        <h2 className="text-2xl font-semibold">Tourist Guardian AI</h2>
        <select 
          value={language} 
          onChange={(e) => setLanguage(e.target.value)}
          className="bg-muted rounded-md px-3 py-1 text-sm"
        >
          <option value="English">English</option>
          <option value="French">French</option>
          <option value="Spanish">Spanish</option>
          <option value="German">German</option>
          <option value="Hindi">Hindi</option>
        </select>
      </div>
      
      <div className="flex-1 overflow-y-auto mb-4 space-y-4">
        {messages.map(message => (
          <div
            key={message.id}
            className={`flex ${message.sender === 'user' ? 'justify-end' : 'justify-start'}`}
          >
            <div
              className={`max-w-[80%] rounded-lg p-3 ${
                message.sender === 'user'
                  ? 'bg-primary text-primary-foreground'
                  : 'bg-muted'
              }`}
            >
              {message.text}
              {message.sender === 'ai' && message.text.includes('directions') && (
                <div className="mt-2 flex space-x-2">
                  <Button size="sm" variant={message.sender === 'ai' ? 'default' : 'outline'}>

                  <Button size="sm" variant="outline">

                  </Button>
                </div>
              )}
            </div>
          </div>
        ))}
      </div>
      
      <div className="flex items-center border border-border rounded-lg overflow-hidden">
        <Input
          value={inputText}
